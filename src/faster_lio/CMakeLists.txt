cmake_minimum_required(VERSION 3.16)
project(faster_lio LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters)
find_package(TBB REQUIRED)
find_package(glog REQUIRED)
find_package(Threads REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/Pose6D.msg")
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} "rosidl_typesupport_cpp")
option(WITH_IVOX_NODE_TYPE_PHC "Use PHC ivox nodes" OFF)
add_library(faster_lio_core src/laser_mapping.cc src/pointcloud_preprocess.cc src/options.cc src/utils.cc)
target_include_directories(faster_lio_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> ${PCL_INCLUDE_DIRS})
ament_target_dependencies(faster_lio_core rclcpp sensor_msgs nav_msgs geometry_msgs tf2_ros pcl_conversions livox_ros_driver2)
target_link_libraries(faster_lio_core ${typesupport} Eigen3::Eigen ${PCL_LIBRARIES} TBB::tbb glog::glog Threads::Threads)
if(WITH_IVOX_NODE_TYPE_PHC)
  target_compile_definitions(faster_lio_core PUBLIC IVOX_NODE_TYPE_PHC)
endif()
foreach(app run_mapping_online run_mapping_offline)
  add_executable(${app} app/${app}.cc)
  target_link_libraries(${app} faster_lio_core)
  ament_target_dependencies(${app} rosbag2_cpp)
  install(TARGETS ${app} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(TARGETS faster_lio_core DESTINATION lib)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
if(BUILD_TESTING)
  enable_testing()
  add_executable(preprocess_check test/preprocess_check.cc)
  target_link_libraries(preprocess_check faster_lio_core)
  add_test(NAME preprocess_check COMMAND preprocess_check)
endif()
ament_package()
