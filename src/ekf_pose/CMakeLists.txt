cmake_minimum_required(VERSION 3.16)
project(ekf)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(ekf src/ekf_node_vio_timesync_with_acc_pub.cpp src/conversion.cpp)
add_executable(imu_add_noise_quat src/imu_add_noise.cpp src/conversion.cpp)
foreach(target ekf imu_add_noise_quat)
  target_include_directories(${target} PRIVATE include)
  target_link_libraries(${target} Eigen3::Eigen)
  target_compile_definitions(${target} PRIVATE EIGEN_DONT_PARALLELIZE)
  ament_target_dependencies(${target} rclcpp sensor_msgs nav_msgs geometry_msgs)
  install(TARGETS ${target} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
